const obj = {
 value: 42,
 reg() { return function() { return this.value; }; },
 arr() { return () => this.value; }
};
const a = obj.reg();
const b = obj.arr();
console.log(a()); // 1
console.log(b()); // 2
console.log(a.call({ value: 7 })); // 3
console.log(b.call({ value: 7 })); // 4
